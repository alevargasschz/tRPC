/**
 * Ejemplo de cliente tRPC
 * Para usar este archivo en un proyecto cliente (Next.js, React, etc.)
 * 
 * 1. Instala las dependencias:
 *    npm install @trpc/client @trpc/server zod
 * 
 * 2. Copia el tipo AppRouter del servidor:
 *    src/trpc/routers/_app.ts -> types/trpc.ts
 */

import { createTRPCProxyClient, httpBatchLink } from '@trpc/client';
import type { AppRouter } from '../features/tasks/trpc/routers/_app';

// Crear cliente tRPC
export const trpc = createTRPCProxyClient<AppRouter>({
  links: [
    httpBatchLink({
      url: 'http://localhost:3000/trpc',
      // Agregar headers personalizados si es necesario
      headers() {
        return {
          'Content-Type': 'application/json',
        };
      },
    }),
  ],
});

/**
 * Ejemplos de uso
 */

// ============== CREAR TAREA ==============
async function crearTarea() {
  try {
    const newTask = await trpc.tasks.create.mutate({
      title: 'Implementar tRPC',
      description: 'Configurar tRPC en el servidor',
      priority: 'Alta',
      isCompleted: false,
    });
    console.log('Tarea creada:', newTask);
    return newTask;
  } catch (error) {
    console.error('Error al crear tarea:', error);
  }
}

// ============== OBTENER TODAS LAS TAREAS ==============
async function obtenerTodasLasTareas() {
  try {
    const tasks = await trpc.tasks.getAll.query();
    console.log('Todas las tareas:', tasks);
    return tasks;
  } catch (error) {
    console.error('Error al obtener tareas:', error);
  }
}

// ============== OBTENER TAREA POR ID ==============
async function obtenerTareaPorId(id: string) {
  try {
    const task = await trpc.tasks.getById.query({ id });
    console.log('Tarea encontrada:', task);
    return task;
  } catch (error) {
    console.error('Error al obtener tarea:', error);
  }
}

// ============== ACTUALIZAR TAREA ==============
async function actualizarTarea(id: string) {
  try {
    const updatedTask = await trpc.tasks.update.mutate({
      id,
      data: {
        isCompleted: true,
        description: 'Tarea actualizada',
      },
    });
    console.log('Tarea actualizada:', updatedTask);
    return updatedTask;
  } catch (error) {
    console.error('Error al actualizar tarea:', error);
  }
}

// ============== ELIMINAR TAREA ==============
async function eliminarTarea(id: string) {
  try {
    const result = await trpc.tasks.delete.mutate({ id });
    console.log('Tarea eliminada:', result);
    return result;
  } catch (error) {
    console.error('Error al eliminar tarea:', error);
  }
}

/**
 * USAR EN REACT
 */
export function EjemploReact() {
  const [tasks, setTasks] = React.useState([]);
  const [loading, setLoading] = React.useState(false);

  const cargarTareas = async () => {
    setLoading(true);
    try {
      const response = await trpc.tasks.getAll.query();
      setTasks(response.data || []);
    } catch (error) {
      console.error('Error:', error);
    } finally {
      setLoading(false);
    }
  };

  const crearNuevaTarea = async (titulo: string, descripcion: string) => {
    try {
      const nuevaTarea = await trpc.tasks.create.mutate({
        title: titulo,
        description: descripcion,
        priority: 'Media',
      });
      setTasks([...tasks, nuevaTarea.data]);
    } catch (error) {
      console.error('Error:', error);
    }
  };

  React.useEffect(() => {
    cargarTareas();
  }, []);

  return (
    <div>
      <h1>Mis Tareas</h1>
      {loading ? <p>Cargando...</p> : null}
      <ul>
        {tasks.map((task: any) => (
          <li key={task._id}>{task.title}</li>
        ))}
      </ul>
      <button onClick={() => crearNuevaTarea('Nueva tarea', 'Descripción')}>
        Agregar tarea
      </button>
    </div>
  );
}

/**
 * USAR EN NEXT.JS CON SERVIDOR
 */
// hooks/useTasks.ts
export function useTasks() {
  const [tasks, setTasks] = React.useState([]);
  const [error, setError] = React.useState<string | null>(null);
  const [loading, setLoading] = React.useState(false);

  const query = async () => {
    setLoading(true);
    try {
      const response = await trpc.tasks.getAll.query();
      setTasks(response.data || []);
      setError(null);
    } catch (err) {
      setError((err as Error).message);
    } finally {
      setLoading(false);
    }
  };

  const create = async (title: string, description: string, priority?: string) => {
    try {
      const response = await trpc.tasks.create.mutate({
        title,
        description,
        priority: (priority || 'Media') as 'Baja' | 'Media' | 'Alta',
      });
      return response.data;
    } catch (err) {
      setError((err as Error).message);
      throw err;
    }
  };

  const update = async (id: string, updates: Partial<{ isCompleted: boolean; description: string }>) => {
    try {
      const response = await trpc.tasks.update.mutate({
        id,
        data: updates,
      });
      return response.data;
    } catch (err) {
      setError((err as Error).message);
      throw err;
    }
  };

  const remove = async (id: string) => {
    try {
      await trpc.tasks.delete.mutate({ id });
    } catch (err) {
      setError((err as Error).message);
      throw err;
    }
  };

  return { tasks, error, loading, query, create, update, remove };
}
